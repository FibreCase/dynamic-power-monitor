# Recompress when the single-file page changes, then embed the generated bytes.
set(WEB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../web/index.html")
set(WEB_GZIP "${CMAKE_CURRENT_BINARY_DIR}/control_page.html.gz")
add_custom_command(OUTPUT "${WEB_GZIP}"
    COMMAND "${PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/../web/compress.py"
            "${WEB_SOURCE}" "${WEB_GZIP}"
    DEPENDS "${WEB_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/../web/compress.py"
    VERBATIM)
add_custom_target(control_page_gzip DEPENDS "${WEB_GZIP}")

add_dependencies(${COMPONENT_LIB} control_page_gzip)
target_add_binary_data(${COMPONENT_LIB} "${WEB_GZIP}" BINARY)
